    Flash Card App: Setup Steps

    Builds the flash card app from an empty folder to the final TypeScript project. Requires Node 18+ and npm.

    Flash card app (TypeScript)

    1. Create the folder and initialize git

    mkdir flashCard && cd flashCard
git init

    2. Get the source code in the terminal

    Connect the folder to the GitHub repo and pull the app files:

    git remote add origin https://github.com/sourabhrustagi/flashCardReact.git
git pull origin main

    This gives you the finished project. Steps 3 to 6 explain how to install, run and check it. Step 7 lists every file to create, in order, with its code.

    3. Install dependencies in the terminal

    If you pulled the repo, one command installs everything listed in package.json:

    npm install

    To build the same dependency list yourself, install the runtime packages and then the dev packages, with pinned versions:

    npm install --save-exact react@18.3.1 react-dom@18.3.1
npm install --save-exact -D vite@5.4.11 @vitejs/plugin-react@4.3.4 typescript@5.6.3 @types/react@18.3.12 @types/react-dom@18.3.1

    4. Run the app

    npm run dev

    Or run ./run.sh, which installs dependencies on first run and then starts the dev server. Open the URL Vite prints, usually http://localhost:5173.

    5. Type-check and build for production

    npm run build

    The build runs tsc --noEmit first, so type errors fail the build. Then Vite writes the output to dist/. Preview it with npm run preview.

    6. Alternative: start from the Vite TypeScript template

    If you'd rather start from Vite's own template and then copy the app files over, use:

    npm create vite@latest flash-card -- --template react-ts
cd flash-card
npm install

    Then replace the template's src/ with the files from step 7 and delete the template's sample files.

    7. Create the files one by one

    Run the commands for files 1 to 3 in order, in the project folder. Then create files 4 onward at the path shown and paste the code exactly. The order keeps imports working.

    mkdir -p src/types src/data src/hooks src/components

    3. Create tsconfig.json

    strict turns on full type checking. noEmit means TypeScript only checks types; Vite does the compiling.

    cat > tsconfig.json <<'EOF'
{
  "compilerOptions": {
    "target": "ES2020",
    "useDefineForClassFields": true,
    "lib": ["ES2020", "DOM", "DOM.Iterable"],
    "module": "ESNext",
    "moduleResolution": "bundler",
    "jsx": "react-jsx",
    "skipLibCheck": true,
    "isolatedModules": true,
    "noEmit": true,
    "strict": true,
    "noUnusedLocals": true,
    "noUnusedParameters": true,
    "noFallthroughCasesInSwitch": true
  },
  "include": ["src"]
}
EOF

    8. src/hooks/flashcardsReducer.ts: pure state transitions

    import { cards } from '../data/cards';

export interface FlashcardsState {
  index: number;
  flipped: boolean;
}

export type FlashcardsAction = { type: 'next' } | { type: 'previous' } | { type: 'flip' };

export const initialFlashcardsState: FlashcardsState = { index: 0, flipped: false };

// Pure: same state and action always give the same result, so it is easy to test.
export function flashcardsReducer(state: FlashcardsState, action: FlashcardsAction): FlashcardsState {
  const last = cards.length - 1;

  switch (action.type) {
    case 'next':
      if (state.index >= last) return state;
      // Moving to another card always shows its question side first.
      return { index: state.index + 1, flipped: false };
    case 'previous':
      if (state.index <= 0) return state;
      return { index: state.index - 1, flipped: false };
    case 'flip':
      return { ...state, flipped: !state.flipped };
  }
}

    9. src/hooks/useFlashcards.ts: connects the reducer to components

    import { useReducer } from 'react';
import { cards } from '../data/cards';
import type { Card } from '../types/card';
import { flashcardsReducer, initialFlashcardsState } from './flashcardsReducer';

export interface FlashcardsView {
  card: Card;
  index: number;
  total: number;
  flipped: boolean;
  isFirst: boolean;
  isLast: boolean;
  next: () => void;
  previous: () => void;
  flip: () => void;
}

// Components read values and call actions. All transitions live in flashcardsReducer.
export function useFlashcards(): FlashcardsView {
  const [state, dispatch] = useReducer(flashcardsReducer, initialFlashcardsState);
  const total = cards.length;

  return {
    card: cards[state.index],
    index: state.index,
    total,
    flipped: state.flipped,
    isFirst: state.index === 0,
    isLast: state.index === total - 1,
    next: () => dispatch({ type: 'next' }),
    previous: () => dispatch({ type: 'previous' }),
    flip: () => dispatch({ type: 'flip' }),
  };
}

    10. src/components/FlashCard.tsx: one card that flips on click

    import type { Card } from '../types/card';

interface FlashCardProps {
  card: Card;
  flipped: boolean;
  onFlip: () => void;
}

export default function FlashCard({ card, flipped, onFlip }: FlashCardProps) {
  return (
    <button
      type="button"
      className={`flash-card ${flipped ? 'is-flipped' : ''}`}
      onClick={onFlip}
      aria-label={flipped ? 'Showing answer, click to show question' : 'Showing question, click to reveal answer'}
    >
      <div className="flash-card-inner">
        <div className="flash-card-face flash-card-front">
          <span className="flash-card-label">Question</span>
          <p>{card.question}</p>
        </div>
        <div className="flash-card-face flash-card-back">
          <span className="flash-card-label">Answer</span>
          <p>{card.answer}</p>
        </div>
      </div>
    </button>
  );
}

    11. src/components/ProgressBar.tsx: "Card X of Y" and a fill bar

    interface ProgressBarProps {
  current: number;
  total: number;
}

export default function ProgressBar({ current, total }: ProgressBarProps) {
  const percent = total ? Math.round((current / total) * 100) : 0;

  return (
    <div className="progress">
      <div className="progress-text">
        Card {current} of {total}
      </div>
      <div
        className="progress-track"
        role="progressbar"
        aria-valuemin={0}
        aria-valuemax={100}
        aria-valuenow={percent}
      >
        <div className="progress-fill" style={{ width: `${percent}%` }} />
      </div>
    </div>
  );
}

    12. src/components/Controls.tsx: Previous, Flip and Next buttons

    interface ControlsProps {
  flipped: boolean;
  isFirst: boolean;
  isLast: boolean;
  onPrevious: () => void;
  onFlip: () => void;
  onNext: () => void;
}

export default function Controls({ flipped, isFirst, isLast, onPrevious, onFlip, onNext }: ControlsProps) {
  return (
    <nav className="controls">
      <button onClick={onPrevious} disabled={isFirst}>
        Previous
      </button>
      <button onClick={onFlip}>{flipped ? 'Show question' : 'Flip'}</button>
      <button onClick={onNext} disabled={isLast}>
        Next
      </button>
    </nav>
  );
}

    13. src/App.tsx: page layout that wires the hook to the components

    import { useFlashcards } from './hooks/useFlashcards';
import FlashCard from './components/FlashCard';
import ProgressBar from './components/ProgressBar';
import Controls from './components/Controls';

export default function App() {
  const { card, index, total, flipped, isFirst, isLast, next, previous, flip } = useFlashcards();

  return (
    <main className="app">
      <h1>Flash Cards</h1>

      <ProgressBar current={index + 1} total={total} />

      <FlashCard card={card} flipped={flipped} onFlip={flip} />

      <Controls
        flipped={flipped}
        isFirst={isFirst}
        isLast={isLast}
        onPrevious={previous}
        onFlip={flip}
        onNext={next}
      />
    </main>
  );
}
